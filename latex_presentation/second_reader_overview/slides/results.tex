
\begin{frame}{Main results}
\framesubtitle{Fine-grained measurement helps most when the construct is separable}

\begin{columns}[t]
    \begin{column}{0.46\textwidth}
        {\large \textbf{FeedbackQA}}\\[0.1cm]
        \hrule
        \vspace{0.2cm}
        \textit{Convergent validity}

        \vspace{0.3cm}
        \begin{itemize}
            \item Formative $>$ holistic baselines
            \item Best Spearman's $\rho$: \textbf{0.578}
            \begin{itemize}
                \item Human-human $\rho$: 0.541
            \end{itemize}
            \item Criteria are strongly intercorrelated
        \end{itemize}

        \vspace{0.2cm}
        \textbf{Interpretation:}\\
        Decomposition improves alignment with human ratings, but dimensions are not cleanly separable.
    \end{column}

    \begin{column}{0.46\textwidth}
        {\large \textbf{Barter Deals}}\\[0.1cm]
        \hrule
        \vspace{0.2cm}
        \textit{Predictive validity}

        \vspace{0.3cm}
        \begin{itemize}
            \item Fine-grained formative scores add strong incremental fit
            \item Best $\Delta$BIC: \textbf{+352}
            \item Poisson deviance: \textbf{14.68 $\rightarrow$ 13.23}
        \end{itemize}

        \vspace{0.2cm}
        \textbf{Interpretation:}\\
        Fine-grained criteria capture meaningful variation in creator application behavior and are cleanly separable.
    \end{column}
\end{columns}

\end{frame}
