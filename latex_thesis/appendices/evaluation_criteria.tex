\section{Evaluation Criteria and Rating Anchors}
\label{app:evaluation_criteria}

\subsection{FeedbackQA Criteria}
\label{app:feedbackqa_criteria}

\begin{table}[h]
    \centering
    \caption{Formative dimensions of \textit{Q\&A Answer Quality}}
    \label{tab:feedbackqa_formative_dimensions}
    \begin{tabularx}{\textwidth}{lX}
    \toprule
    Dimension & Definition \\
    \midrule
    \textit{Topical Alignment} & The degree to which the answer addresses the issue and stated constraints of the question. \\
    \textit{Information Coverage} & The degree to which the answer provides enough relevant information to serve as a full answer. \\
    \textit{Topical Concentration} & The degree to which the answer remains centered on the asked issue ("on-topic") throughout the passage. \\
    \textit{Communicative Clarity} & The degree to which the answer is expressed in a clear, well-organized, and easy-to-follow way. \\
    \bottomrule
    \end{tabularx}
\end{table}

\subsection{Barter formative dimensions}
\label{app:barter_criteria}

\begin{table}[!htbp]
    \centering
    \footnotesize
    \setlength{\tabcolsep}{4pt}
    \renewcommand{\arraystretch}{1.08}
    \caption{Fine-grained dimensions of \textit{Deal Attractiveness} mapped to the macro-formative pillars}
    \label{tab:barter_formative_dimensions}
    \begin{tabularx}{\textwidth}{@{}
        >{\raggedright\arraybackslash}p{0.24\textwidth}
        >{\raggedright\arraybackslash}p{0.16\textwidth}
        >{\raggedright\arraybackslash}X
    @{}}
    \toprule
    Dimension & Pillar & Definition \\
    \midrule
    \textit{Brand Prestige} & Reward & How well-known and trusted the brand name is in the market. \\
    \textit{Perceived Economic Value} & Reward & The estimated retail value and premium nature of the product, service, or compensation offered to the creator. \\
    \textit{Creator Brand Alignment} & Reward & How naturally the product, service, or experience fits within a professional content creator's social media presence. \\
    \textit{Functional Utility} & Reward & The practical, everyday usefulness and utilitarian value of the physical product or service being offered. \\
    \textit{Product Universality} & Reward & The mass-market consumer appeal of the physical product or service on offer. \\
    \midrule
    \textit{Workload Volume} & Cost & The total time and effort required to execute the collaboration. \\
    \textit{Creative Restrictiveness} & Cost & The level of brand control over the creator's creative process. \\
    \midrule
    \textit{Call-to-Action Strength} & Pitch Quality & The level of excitement and warmth in the brand's invitation to collaborate. \\
    \textit{Rhetorical Objectivity} & Pitch Quality & The level of factual realism versus promotional marketing hype in the pitch. \\
    \textit{Proposition Clarity} & Pitch Quality & The structural clarity and cognitive ease of processing the pitch. \\
    \bottomrule
    \end{tabularx}
\end{table}
